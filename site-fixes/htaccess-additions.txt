# =====================================================================
#  TheGreensolve SEO fixes for public_html/.htaccess
#  1. Download and keep a backup copy of your current .htaccess first.
#  2. Paste SECTION A at the very top of the file.
#  3. Find the catch-all rule (SECTION B explains what it looks like)
#     and replace it with SECTION C.
#  4. Save, then run the checks at the bottom.
# =====================================================================

# ---------- SECTION A: paste at the top ----------
Options -MultiViews
RewriteEngine On

# www -> https://thegreensolve.com in one hop
RewriteCond %{HTTP_HOST} ^www\. [NC]
RewriteRule ^ https://thegreensolve.com%{REQUEST_URI} [R=301,L]

# /index.html -> /   and   /blog/index.html -> /blog  (removes duplicate URLs)
RewriteCond %{THE_REQUEST} \s/[^\s?]*index\.html[\s?] [NC]
RewriteRule ^(.*?)/?index\.html$ /$1 [R=301,L]

# Security headers
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000"
  Header always set X-Frame-Options "SAMEORIGIN"
</IfModule>

# ---------- SECTION B: DELETE the existing catch-all ----------
# It sends every missing URL to the homepage, which causes the soft 404s.
# It usually looks like this (yours may differ slightly):
#
#   RewriteCond %{REQUEST_FILENAME} !-f
#   RewriteCond %{REQUEST_FILENAME} !-d
#   RewriteRule . /index.html [L]
#
# Hostinger may also have it as:  FallbackResource /index.html
# Remove whichever one is there.

# ---------- SECTION C: put this where the catch-all was ----------
# Net-Low is a single-page app, so only its routes fall back to its own shell
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^netlow(/.*)?$ /netlow/index.html [L]

# Everything else that doesn't exist gets a real 404
ErrorDocument 404 /404.html

#  Checks after saving (run in a terminal, or open the URLs in a browser)

#    curl -s -o /dev/null -w "%{http_code}\n" https://thegreensolve.com/blog                -> 200
#    curl -s -o /dev/null -w "%{http_code}\n" https://thegreensolve.com/netlow/documents    -> 200
#    curl -sI https://thegreensolve.com/index.html | grep -i location                     -> https://thegreensolve.com/
#  If the site shows an error after saving, put the backup .htaccess back.
